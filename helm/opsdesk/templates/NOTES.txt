  kubectl -n {{ .Release.Namespace }} get pods -l app.kubernetes.io/instance={{ .Release.Name }}

  kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "opsdesk.fullname" . }}-api 8080:{{ .Values.api.service.port }}
